use async_trait::async_trait;
use pingora::prelude::*;
use std::path::{Path, PathBuf};
use tokio::fs;

pub struct FileServer {
    root_dir: PathBuf,
}

impl FileServer {
    pub fn new<P: AsRef<Path>>(path: P) -> Self {
        Self {
            root_dir: path.as_ref().to_path_buf(),
        }
    }
}

#[async_trait]
impl ProxyHttp for FileServer {
    type CTX = ();

    fn new_ctx(&self) -> Self::CTX {}

    /// Intercept incoming requests and serve static files directly from disk.
    async fn request_filter(
        &self,
        session: &mut Session,
        _ctx: &mut Self::CTX,
    ) -> Result<bool> {
        let req = session.req_header();
        let path_str = req.uri.path().trim_start_matches('/');

        // Sanitize path to prevent directory traversal attacks (e.g. ../../etc/passwd)
        let requested_path = self.root_dir.join(path_str);
        if !requested_path.starts_with(&self.root_dir) {
            let mut header = ResponseHeader::build(403, None).unwrap();
            header.insert_header("Content-Type", "text/plain").unwrap();
            session.write_response_header(Box::new(header), true).await?;
            return Ok(true); // Return true to signal that Pingora has finished handling the response
        }

        // Handle root or directory requests by defaulting to index.html
        let target_path = if requested_path.is_dir() {
            requested_path.join("index.html")
        } else {
            requested_path
        };

        // Attempt to read the file from disk
        match fs::read(&target_path).await {
            Ok(content) => {
                // Guess MIME type based on file extension
                let mime_type = mime_guess::from_path(&target_path)
                    .first_or_octet_stream()
                    .to_string();

                let mut header = ResponseHeader::build(200, None).unwrap();
                header.insert_header("Content-Type", mime_type).unwrap();
                header
                    .insert_header("Content-Length", content.len().to_string())
                    .unwrap();

                // Send headers and body back to client
                session.write_response_header(Box::new(header), false).await?;
                session.write_response_body(Some(content.into()), true).await?;

                println!("==> Served [200]: {}", target_path.display());
            }
            Err(_) => {
                // Return 404 Not Found if file does not exist
                let mut header = ResponseHeader::build(404, None).unwrap();
                header.insert_header("Content-Type", "text/plain").unwrap();
                
                session.write_response_header(Box::new(header), false).await?;
                session
                    .write_response_body(Some("404 Not Found".into()), true)
                    .await?;

                println!("==> Not Found [404]: {}", target_path.display());
            }
        }

        // Return `true` to indicate the response was fully handled in this filter
        Ok(true)
    }

    /// Required by ProxyHttp trait, but unreachable because request_filter handles all requests.
    async fn upstream_peer(
        &self,
        _session: &mut Session,
        _ctx: &mut Self::CTX,
    ) -> Result<Box<HttpPeer>> {
        Err(Error::explain(
            ErrorType::HTTPStatus(500),
            "Static file server does not use upstream peers",
        ))
    }
}

fn main() {
    env_logger::init_from_env(env_logger::Env::default().default_filter_or("info"));

    let mut my_server = Server::new(None).unwrap();
    my_server.bootstrap();

    // Instantiate FileServer pointing to /var/www/repos
    let static_dir = "/var/www/repos";
    let file_service = FileServer::new(static_dir);

    // Create the HTTP proxy service and bind to port 6647
    let mut service = http_proxy_service(&my_server.configuration, file_service);
    service.add_tcp("0.0.0.0:6647");

    println!("Serving files from {} on http://0.0.0.0:6647", static_dir);

    my_server.add_service(service);
    my_server.run_forever();
}
