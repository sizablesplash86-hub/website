use async_trait::async_trait;
use pingora::prelude::*;
use std::sync::Arc;

/// A wrapper struct for our LoadBalancer instance.
pub struct LB(Arc<LoadBalancer<RoundRobin>>);

#[async_trait]
impl ProxyHttp for LB {
    type CTX = ();

    fn new_ctx(&self) -> Self::CTX {}

    /// Intercepts incoming requests right when they hit the proxy to log method and URI.
    async fn request_filter(
        &self,
        session: &mut Session,
        _ctx: &mut Self::CTX,
    ) -> Result<bool> {
        let req = session.req_header();
        println!("==> Proxy intercepted: {} {}", req.method, req.uri);
        // Returning Ok(false) signals Pingora to continue processing the request normally.
        Ok(false)
    }

    /// Selects an upstream peer using round-robin selection across active healthy backends.
    async fn upstream_peer(
        &self,
        _session: &mut Session,
        _ctx: &mut Self::CTX,
    ) -> Result<Box<HttpPeer>> {
        let upstream = self
            .0
            .select(b"", 256) // Hash key is unused for round-robin
            .unwrap();

        println!("upstream peer is: {upstream:?}");

        // Constructs the HttpPeer.
        // Sets `tls` to `true` and passes the SNI for HTTPS upstreams.
        let peer = Box::new(HttpPeer::new(
            upstream,
            true,
            "one.one.one.one".to_string(),
        ));
        Ok(peer)
    }

    /// Modifies request headers sent to the upstream server.
    async fn upstream_request_filter(
        &self,
        _session: &mut Session,
        upstream_request: &mut RequestHeader,
        _ctx: &mut Self::CTX,
    ) -> Result<()> {
        upstream_request
            .insert_header("Host", "one.one.one.one")
            .unwrap();
        Ok(())
    }
}

fn main() {
    // Initialize logger environment
    env_logger::init_from_env(env_logger::Env::default().default_filter_or("info"));

    // Read and parse CLI arguments (e.g., -c conf.yaml, -d for daemonize, -u for graceful upgrade)
//  let opt = Opt::parse_args();
    let mut my_server = Server::new(None).unwrap();
    my_server.bootstrap();

    // Define upstream backends (includes healthy backends and an offline backend to test health checks)
    let mut upstreams = LoadBalancer::try_from_iter([
        "1.1.1.1:443",
        "1.0.0.1:443",
//        "127.0.0.1:343", // Intentionally offline server
    ])
    .unwrap();

    // Set up TCP health checks for backends
    let hc = TcpHealthCheck::new();
    upstreams.set_health_check(hc);
    upstreams.health_check_frequency = Some(std::time::Duration::from_secs(1));

    // Wrap the load balancer in a background service for continuous health checking
    let background = background_service("health check", upstreams);
    let upstreams = background.task();

    // Initialize the HTTP proxy service listening locally on port 6188
    let mut lb = http_proxy_service(&my_server.configuration, LB(upstreams));
    lb.add_tcp("0.0.0.0:6188");

    // Add background health checker and proxy service to the Pingora server
    my_server.add_service(background);
    my_server.add_service(lb);

    // Run the Pingora server runtime
    my_server.run_forever();
}
