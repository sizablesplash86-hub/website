use async_trait::async_trait;
use pingora::prelude::*;
use std::sync::Arc;

/// A wrapper struct for our LoadBalancer instance.
pub struct LB(Arc<LoadBalancer<RoundRobin>>);

#[async_trait]
impl ProxyHttp for LB {
    /// Context storage is not needed for this load balancer example.
    type CTX = ();

    fn new_ctx(&self) -> () {
        ()
    }

    /// Selects an upstream peer using the round-robin selection algorithm.
    async fn upstream_peer(&self, _session: &mut Session, _ctx: &mut ()) -> Result<Box<HttpPeer>> {
        let upstream = self
            .0
            .select(b"", 256) // Hash key is unused for round-robin selection
            .unwrap();

        println!("upstream peer is: {upstream:?}");

        // Constructs the HttpPeer.
        // - `true` enables TLS for HTTPS backends (set to `false` for plain HTTP).
        // - `"one.one.one.one"` sets the SNI (pass an empty string `""` for plain HTTP).
        let peer = Box::new(HttpPeer::new(
            upstream,
            true,
            "one.one.one.one".to_string(),
        ));
        Ok(peer)
    }

    /// Modifies the request headers sent to the upstream backend.
    async fn upstream_request_filter(
        &self,
        _session: &mut Session,
        upstream_request: &mut RequestHeader,
        _ctx: &mut Self::CTX,
    ) -> Result<()> {
        upstream_request
            .insert_header("Host", "one.one.one.one")
            .unwrap();
        Ok(())
    }
}

fn main() {
    // Read and parse CLI arguments (e.g., -c conf.yaml, -d for daemonize, -u for graceful upgrade)
    let opt = Opt::parse_args();
    let mut my_server = Server::new(Some(opt)).unwrap();
    my_server.bootstrap();

    // Define upstream backends (includes healthy backends and an optional offline backend to demonstrate health checks)
    let mut upstreams = LoadBalancer::try_from_iter([
        "1.1.1.1:443",
        "1.0.0.1:443",
        "127.0.0.1:343", // Intentionally offline server
    ])
    .unwrap();

    // Set up TCP health checks for backends
    let hc = TcpHealthCheck::new();
    upstreams.set_health_check(hc);
    upstreams.health_check_frequency = Some(std::time::Duration::from_secs(1));

    // Wrap the load balancer in a background service to continuously run health checks
    let background = background_service("health check", upstreams);
    let upstreams = background.task();

    // Initialize the HTTP proxy service listening locally on port 6188
    let mut lb = http_proxy_service(&my_server.configuration, LB(upstreams));
    lb.add_tcp("0.0.0.0:6188");

    // Add background health checker and load balancer proxy service to the server
    my_server.add_service(background);
    my_server.add_service(lb);

    // Run the Pingora server runtime
    my_server.run_forever();
}
